// ค้นหาหรือสร้างโฟลเดอร์ย่อยตามชื่อครู
function getOrCreateTeacherFolder(parentFolderId, teacherName) {
  const parentFolder = DriveApp.getFolderById(parentFolderId);
  const folderName = teacherName.trim();
  const folders = parentFolder.getFoldersByName(folderName);
  if (folders.hasNext()) {
    return folders.next();
  }
  return parentFolder.createFolder(folderName);
}

// 1. เข้าสู่ระบบ
function loginUser(username, password) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Users");
    if (!sheet) return { success: false, message: "ไม่พบแท็บ Users ในฐานข้อมูล" };

    const data = sheet.getDataRange().getDisplayValues();
    for (let i = 1; i < data.length; i++) {
      const u = data[i][0].trim();
      const p = data[i][1].trim();
      const role = data[i][2].trim();
      const fullName = data[i][3].trim();
      const schoolName = data[i][4].trim();
      const subjectGroup = data[i][5].trim();
      const email = data[i][6].trim();
      const status = data[i][7].trim();

      if (u === String(username).trim() && p === String(password).trim()) {
        if (status.toLowerCase() === "inactive") {
          return { success: false, message: "บัญชีนี้ถูกระงับการใช้งานชั่วคราว" };
        }
        return {
          success: true,
          user: {
            username: u,
            role: role,
            fullName: fullName,
            schoolName: schoolName,
            subjectGroup: subjectGroup,
            email: email,
            isDefaultPass: (p === "1234")
          }
        };
      }
    }
    return { success: false, message: "ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง" };
  } catch (err) {
    return { success: false, message: "Error: " + err.toString() };
  }
}

// 2. ปรับปรุงข้อมูลส่วนตัว & รหัสผ่าน
function updateUserProfile(payload) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Users");
    if (!sheet) return { success: false, message: "ไม่พบแท็บ Users" };

    const data = sheet.getDataRange().getDisplayValues();
    const currentUsername = String(payload.currentUsername).trim();
    let targetRow = -1;

    for (let i = 1; i < data.length; i++) {
      const existingUser = data[i][0].trim();
      if (existingUser === currentUsername) {
        targetRow = i + 1;
      } else if (existingUser === String(payload.newUsername).trim()) {
        return { success: false, message: "ชื่อผู้ใช้งาน (Username) นี้มีผู้ใช้งานแล้ว โปรดใช้ชื่ออื่น" };
      }
    }

    if (targetRow === -1) {
      return { success: false, message: "ไม่พบบัญชีผู้ใช้งานในระบบ" };
    }

    sheet.getRange(targetRow, 1, 1, 8).setNumberFormat("@");
    sheet.getRange(targetRow, 1).setValue(payload.newUsername.trim());
    if (payload.newPassword && payload.newPassword.trim() !== "") {
      sheet.getRange(targetRow, 2).setValue(payload.newPassword.trim());
    }
    sheet.getRange(targetRow, 4).setValue(payload.fullName.trim());
    sheet.getRange(targetRow, 5).setValue(payload.schoolName.trim());
    sheet.getRange(targetRow, 6).setValue(payload.subjectGroup.trim());
    sheet.getRange(targetRow, 7).setValue(payload.email.trim());

    return {
      success: true,
      message: "ปรับปรุงข้อมูลส่วนตัวเรียบร้อยแล้ว",
      updatedUser: {
        username: payload.newUsername.trim(),
        fullName: payload.fullName.trim(),
        schoolName: payload.schoolName.trim(),
        subjectGroup: payload.subjectGroup.trim(),
        email: payload.email.trim()
      }
    };
  } catch (err) {
    return { success: false, message: "เกิดข้อผิดพลาด: " + err.toString() };
  } finally {
    lock.releaseLock();
  }
}

// 3. ดึงข้อมูลระบบ
function getInitialData(username, role) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    // ภารกิจ (Assignments)
    const assignSheet = ss.getSheetByName("Assignments");
    let assignments = [];
    if (assignSheet && assignSheet.getLastRow() > 1) {
      const aData = assignSheet.getDataRange().getDisplayValues();
      for (let i = 1; i < aData.length; i++) {
        assignments.push({
          id: aData[i][0],
          title: aData[i][1],
          description: aData[i][2],
          dueDate: aData[i][3],
          templateLink: aData[i][4],
          status: aData[i][5]
        });
      }
    }

    // ประวัติการส่งงาน (Submissions)
    const subSheet = ss.getSheetByName("Submissions");
    let submissions = [];
    if (subSheet && subSheet.getLastRow() > 1) {
      const sData = subSheet.getDataRange().getDisplayValues();
      for (let i = 1; i < sData.length; i++) {
        if (role === "teacher" && String(sData[i][2]).trim() !== String(username).trim()) {
          continue;
        }
        submissions.push({
          rowId: i + 1,
          submissionId: sData[i][0],
          timestamp: sData[i][1] || "-",
          username: sData[i][2],
          teacherName: sData[i][3],
          schoolName: sData[i][4],
          assignmentId: sData[i][5],
          researchTitle: sData[i][6],
          fileLink: sData[i][7],
          teacherReflection: sData[i][8],
          status: sData[i][9],
          supervisorComment: sData[i][10],
          feedbackDate: sData[i][11] || "-"
        });
      }
    }

    // คลังสื่อ (Resources)
    const resSheet = ss.getSheetByName("Resources");
    let resources = [];
    if (resSheet && resSheet.getLastRow() > 1) {
      const rData = resSheet.getDataRange().getDisplayValues();
      for (let i = 1; i < rData.length; i++) {
        resources.push({
          id: rData[i][0],
          category: rData[i][1],
          title: rData[i][2],
          description: rData[i][3],
          fileLink: rData[i][4],
          thumbnailUrl: rData[i][5],
          updatedDate: rData[i][6]
        });
      }
    }

    // กระดานถามตอบ (QA_Board)
    const qaSheet = ss.getSheetByName("QA_Board");
    let qaBoard = [];
    if (qaSheet && qaSheet.getLastRow() > 1) {
      const qData = qaSheet.getDataRange().getDisplayValues();
      for (let i = 1; i < qData.length; i++) {
        qaBoard.push({
          rowId: i + 1,
          postId: qData[i][0],
          timestamp: qData[i][1] || "-",
          username: qData[i][2],
          teacherName: qData[i][3],
          schoolName: qData[i][4],
          topic: qData[i][5],
          detail: qData[i][6],
          answer: qData[i][7],
          answeredBy: qData[i][8],
          status: qData[i][9]
        });
      }
    }

    return {
      success: true,
      assignments: assignments,
      submissions: submissions.reverse(),
      resources: resources,
      qaBoard: qaBoard.reverse()
    };
  } catch (err) {
    return { success: false, message: "Error: " + err.toString() };
  }
}

// 5. เพิ่มสื่อลงคลัง Resources

  lock.tryLock(15000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Resources");
    if (!sheet) return { success: false, message: "ไม่พบแท็บ Resources ในฐานข้อมูล" };

    const todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy");
    let resId = (payload.resource_id || "").trim();
    if (!resId) {
      resId = "RES-" + Utilities.formatDate(new Date(), "GMT+7", "yyMMddHHmmss");
    }

    const category = (payload.category || "ทั่วไป").trim();
    const title = (payload.title || "สื่อการเรียนรู้ Jamovi").trim();
    const description = (payload.description || "-").trim();
    const fileLink = (payload.file_link || "").trim();
    const thumbnailUrl = (payload.thumbnail_url || "").trim();
    const updatedDate = (payload.updated_date || todayStr).trim();

    sheet.appendRow([resId, category, title, description, fileLink, thumbnailUrl, updatedDate]);

    return { success: true, message: "บันทึกสื่อและคู่มือลงในคลังเรียบร้อยแล้ว" };
  } catch (err) {
    return { success: false, message: "เกิดข้อผิดพลาดในการบันทึกสื่อ: " + err.toString() };
  } finally {
    lock.releaseLock();
  }
}

// 6. ส่งงานภารกิจ
function submitAssignment(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Submissions");
    if (!sheet) return { success: false, message: "ไม่พบแท็บ Submissions" };

    const teacherFolder = getOrCreateTeacherFolder(SUBMISSION_FOLDER_ID, payload.teacherName);
    let fileUrl = payload.fileLink || "";

    if (payload.fileData && payload.fileData.includes("base64,")) {
      const parts = payload.fileData.split("base64,");
      const contentType = parts[0].split(":")[1].split(";")[0];
      const decodedBytes = Utilities.base64Decode(parts[1]);
      const blob = Utilities.newBlob(decodedBytes, contentType, payload.fileName);
      const savedFile = teacherFolder.createFile(blob);
      savedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fileUrl = savedFile.getUrl();
    }

    const submissionId = "SUB-" + Utilities.formatDate(new Date(), "GMT+7", "yyMMddHHmmss");
    const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm");

    sheet.appendRow([
      submissionId,
      timestamp,
      payload.username,
      payload.teacherName,
      payload.schoolName,
      payload.assignmentId,
      payload.researchTitle,
      fileUrl,
      payload.reflection,
      "รอตรวจ",
      "",
      ""
    ]);

    return { success: true, message: "ส่งงานเรียบร้อยแล้ว", fileUrl: fileUrl };
  } catch (err) {
    return { success: false, message: "ส่งงานล้มเหลว: " + err.toString() };
  }
}

// 7. ส่งงานฉบับแก้ไข
function resubmitAssignment(payload) {
  const lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Submissions");
    if (!sheet) return { success: false, message: "ไม่พบแท็บ Submissions" };

    const row = Number(payload.rowId);
    let fileUrl = payload.fileLink || "";

    if (payload.fileData && payload.fileData.includes("base64,")) {
      const teacherFolder = getOrCreateTeacherFolder(SUBMISSION_FOLDER_ID, payload.teacherName);
      const parts = payload.fileData.split("base64,");
      const contentType = parts[0].split(":")[1].split(";")[0];
      const decodedBytes = Utilities.base64Decode(parts[1]);
      const blob = Utilities.newBlob(decodedBytes, contentType, "แก้ไข_" + payload.fileName);
      const savedFile = teacherFolder.createFile(blob);
      savedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fileUrl = savedFile.getUrl();
    }

    const nowStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm");

    sheet.getRange(row, 2).setValue(nowStr + " (ส่งฉบับแก้ไข)");
    sheet.getRange(row, 7).setValue(payload.researchTitle);
    if (fileUrl) {
      sheet.getRange(row, 8).setValue(fileUrl);
    }
    sheet.getRange(row, 9).setValue(payload.reflection);
    sheet.getRange(row, 10).setValue("รอตรวจ (ส่งแก้ไขแล้ว)");

    return { success: true, message: "ส่งงานฉบับแก้ไขเรียบร้อยแล้ว สถานะถูกปรับเป็นรอตรวจ" };
  } catch (err) {
    return { success: false, message: "เกิดข้อผิดพลาดในการส่งงานแก้ไข: " + err.toString() };
  } finally {
    lock.releaseLock();
  }
}
