    function handleLogin(e) {
      e.preventDefault();
      const u = document.getElementById('loginUser').value.trim();
      const p = document.getElementById('loginPass').value.trim();
      const btn = document.getElementById('loginBtn');

      btn.disabled = true;
      btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> กำลังตรวจสอบ...';

        .withSuccessHandler(res => {
          btn.disabled = false;
          btn.innerHTML = '<span>เข้าสู่ระบบ</span> <i class="fa-solid fa-arrow-right"></i>';
          if (res.success) {
            currentUser = res.user;
            localStorage.setItem('jamovi_mentoring_user', JSON.stringify(currentUser));
            showPortal();
          } else {
            Swal.fire({ icon: 'error', title: 'เข้าสู่ระบบไม่สำเร็จ', text: res.message, confirmButtonColor: '#1e3a8a' });

        })
        .loginUser(u, p);
    }

      else btnNewTask.classList.add('hidden');

      const btnNewRes = document.getElementById('btnOpenNewResource');
      if (isManager) btnNewRes.classList.remove('hidden');
      else btnNewRes.classList.add('hidden');

      if (currentUser.isDefaultPass) {
        document.getElementById('defaultPassNotice').classList.remove('hidden');
      } else {
        document.getElementById('defaultPassNotice').classList.add('hidden');
      }

      fillProfileForm();
      loadSystemData();
    }

    function handleAssignmentSubmit(e) {
      e.preventDefault();
      const fileInput = document.getElementById('subFileInput');
      const btn = document.getElementById('btnUploadSubmit');

      btn.disabled = true;
      btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> กำลังอัปโหลด...';

      const payload = {
        username: currentUser.username,
        teacherName: currentUser.fullName,
        schoolName: currentUser.schoolName,
        assignmentId: document.getElementById('subTaskId').value,
        researchTitle: document.getElementById('subResearchTitle').value.trim(),
        fileLink: document.getElementById('subFileLink').value.trim(),
        reflection: document.getElementById('subReflection').value.trim(),
        fileName: '',
        fileData: ''
      };

      if (fileInput.files.length > 0) {
        const file = fileInput.files[0];
        payload.fileName = file.name;
        const reader = new FileReader();
        reader.onload = function(evt) {
          payload.fileData = evt.target.result;
          sendAssignmentData(payload, btn);
        };
        reader.readAsDataURL(file);
      } else {
        sendAssignmentData(payload, btn);
      }
    }

    function sendAssignmentData(payload, btn) {
      google.script.run
        .withSuccessHandler(res => {
          btn.disabled = false;
          btn.innerHTML = '<i class="fa-solid fa-paper-plane"></i> ยืนยันการส่งงาน';
          if (res.success) {
            Swal.fire({ icon: 'success', title: 'ส่งงานเรียบร้อย', text: res.message, confirmButtonColor: '#1e3a8a' });
            closeSubmitForm();
            loadSystemData();
          } else {
            Swal.fire({ icon: 'error', title: 'ล้มเหลว', text: res.message, confirmButtonColor: '#1e3a8a' });
          }
        })
        .withFailureHandler(err => {
          btn.disabled = false;
          btn.innerHTML = '<i class="fa-solid fa-paper-plane"></i> ยืนยันการส่งงาน';
          Swal.fire({ icon: 'error', title: 'Error', text: err.toString(), confirmButtonColor: '#1e3a8a' });
        })
        .submitAssignment(payload);
    }

    function renderSubmissions(list) {
      const tbody = document.getElementById('submissionTableBody');
      if (!list || list.length === 0) {

      }
      tbody.innerHTML = list.map(s => {

        if (s.status === 'ผ่านการนิเทศ') badgeColor = 'bg-emerald-100 text-emerald-800';
        else if (s.status === 'ปรับปรุงแก้ไข') badgeColor = 'bg-rose-100 text-rose-800';
        else if (s.status.includes('ส่งแก้ไข')) badgeColor = 'bg-blue-100 text-blue-800';

        const isTeacher = (currentUser.role === 'teacher');
        const canFeedback = (currentUser.role === 'supervisor' || currentUser.role === 'admin');
        const canResubmit = (isTeacher && (s.status === 'ปรับปรุงแก้ไข' || s.status.includes('รอตรวจ')));

            <td class="px-3 py-2.5 max-w-[180px] font-semibold text-blue-900">${s.researchTitle}</td>
            <td class="px-3 py-2.5">${s.fileLink ? `<a href="${s.fileLink}" target="_blank" class="text-blue-600 hover:underline"><i class="fa-solid fa-arrow-up-right-from-square"></i> เปิดไฟล์</a>` : '-'}</td>
            <td class="px-3 py-2.5"><span class="px-2 py-0.5 rounded text-[11px] font-bold ${badgeColor}">${s.status || 'รอตรวจ'}</span></td>
